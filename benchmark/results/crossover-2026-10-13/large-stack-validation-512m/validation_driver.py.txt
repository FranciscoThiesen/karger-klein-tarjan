def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def inventory():
    return {str(p.relative_to(root)):sha(p) for pattern in ('*/*.cpp','*/*.hpp')
            for p in sorted(root.glob(pattern))}

info = {'schema':'classical_large_stack_validation_v1', 'validation':'running',

        'original_binary_sha256':sha(original),'source_sha256':inventory(),
        'validation_driver_sha256':sha(__file__),'commands':[]}

def save():
    (out/'provenance.json').write_text(json.dumps(info,indent=2)+'\n')

def run(command, name, timeout=180):
    start=time.monotonic()
    result=subprocess.run(command,cwd=root,text=True,capture_output=True,timeout=timeout)
    (out/(name+'.stdout')).write_text(result.stdout)
    (out/(name+'.stderr')).write_text(result.stderr)
    info['commands'].append({'command':command,'working_directory':str(root),
        'exit_status':result.returncode,'operational_elapsed_seconds':time.monotonic()-start,
        'stdout':name+'.stdout','stdout_sha256':sha(out/(name+'.stdout')),
        'stderr':name+'.stderr','stderr_sha256':sha(out/(name+'.stderr'))})
    save()
    assert result.returncode==0, f'{name} exited {result.returncode}'
    return result.stdout

def macho(path):
    data=path.read_bytes()
    magic,_,_,_,count,_,_,_=struct.unpack_from('<IiiIIIII',data)
    assert magic==0xfeedfacf, 'expected native thin little-endian Mach-O64'
    offset=32
    result={}
    for _ in range(count):
        command,length=struct.unpack_from('<II',data,offset)
        assert length>=8 and offset+length<=len(data)
        if command==0x80000028:
            _,_,entry,stack=struct.unpack_from('<IIQQ',data,offset)
            result['entry_file_offset']=entry
            result['main_stack_size']=stack
        elif command==0x19:
            segment=struct.unpack_from('<II16sQQQQiiII',data,offset)
            for index in range(segment[-2]):
                section=struct.unpack_from('<16s16sQQIIIIIIII',data,offset+72+index*80)
                name,seg=section[0].rstrip(b'\0'),section[1].rstrip(b'\0')
                if (seg,name)==(b'__TEXT',b'__text'):
                    address,size,file_offset=section[2:5]
                    body=data[file_offset:file_offset+size]
                    assert len(body)==size
                    result.update(text_address=address,text_size=size,text_file_offset=file_offset,
                                  text_sha256=hashlib.sha256(body).hexdigest())
        offset+=length
    assert 'main_stack_size' in result and 'text_sha256' in result
    return result

#include <cstdint>
#include <iostream>
#include <stdexcept>
#include <tuple>
#include <vector>
int main() {
    try {
        constexpr int n=4194304;
        std::vector<std::tuple<int,int,int,int>> graph;
        graph.reserve(n-1);
        // Increasing weights force every path edge into the first selected
        // forest. Vertex0 is an endpoint, so the next explore(0) visits all n
        // vertices before unwinding. A connected input tree is its unique MST.
        for(int v=1;v<n;++v) graph.emplace_back(v-1,v,v,v-1);
        const auto result=boruvka(graph,n);
        if(result.size()!=graph.size()) throw std::runtime_error("wrong output size");
        std::vector<unsigned char> seen(n-1,0);
        std::uint64_t sum=0;
        for(const auto& edge:result) {
            const auto [u,v,w,id]=edge;
            if(id<0 || id>=n-1 || seen[id]++ || w!=id+1 ||
               !((u==id && v==id+1)||(v==id && u==id+1)))
                throw std::runtime_error("wrong/duplicate original path tuple");
            sum+=static_cast<std::uint64_t>(w);
        }
        const auto expected=static_cast<std::uint64_t>(n)*(n-1)/2;
        if(sum!=expected) throw std::runtime_error("wrong64-bit total weight");
        std::cout << "{\"validation\":\"pass\",\"vertices\":" << n
                  << ",\"edges\":" << n-1 << ",\"validated_output_tuples\":" << result.size()
                  << ",\"weight_sum\":" << sum
                  << ",\"oracle\":\"connected input tree, every original tuple exactly once\","
                     "\"forced_maximum_dfs_depth\":" << n << "}\n";
    } catch(const std::exception& e) { std::cerr<<e.what()<<'\n';return 1; }
}
''')
